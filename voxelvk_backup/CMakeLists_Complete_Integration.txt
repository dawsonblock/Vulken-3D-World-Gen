# Complete VoxelRL_All CMakeLists.txt with full AI integration and fixes
cmake_minimum_required(VERSION 3.24)
project(VoxelRL_All VERSION 1.0.0 LANGUAGES C CXX CUDA)

# Set C++ standard
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)

# Build options
option(VOXELVK_ENABLE_AI_GENERATION "Enable AI content generation features" ON)
option(VOXELVK_ENABLE_TENSORRT "Enable TensorRT AI inference" ON)
option(VOXELVK_ENABLE_AI_TRAINING "Enable AI-enhanced training features" ON)
option(VOXELVK_ENABLE_CUDA "Enable CUDA acceleration" ON)
option(VOXELVK_HEADLESS_ONLY "Build headless version only" OFF)
option(ENABLE_TESTS "Enable test building" ON)

# Set build type if not specified
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

# Compiler flags
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall -Wextra")
set(CMAKE_CXX_FLAGS_DEBUG "-g -O0 -DDEBUG")
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG -march=native")

# CUDA flags
if(VOXELVK_ENABLE_CUDA)
    set(CMAKE_CUDA_FLAGS "${CMAKE_CUDA_FLAGS} --expt-relaxed-constexpr")
    set(CMAKE_CUDA_FLAGS_DEBUG "-g -G -O0")
    set(CMAKE_CUDA_FLAGS_RELEASE "-O3 --use_fast_math")
endif()

# Find required packages
find_package(PkgConfig REQUIRED)

# Vulkan
find_package(Vulkan REQUIRED)
if(NOT Vulkan_FOUND)
    message(FATAL_ERROR "Vulkan not found")
endif()

# GLM
find_package(glm QUIET)
if(NOT glm_FOUND)
    find_path(GLM_INCLUDE_DIR glm/glm.hpp HINTS /usr/include /usr/local/include)
    if(NOT GLM_INCLUDE_DIR)
        message(FATAL_ERROR "GLM not found")
    endif()
endif()

# Python development files
find_package(Python3 COMPONENTS Interpreter Development REQUIRED)

# pybind11
find_package(pybind11 QUIET)
if(NOT pybind11_FOUND)
    # Try to find pybind11 via Python
    execute_process(
        COMMAND ${Python3_EXECUTABLE} -c "import pybind11; print(pybind11.get_cmake_dir())"
        OUTPUT_VARIABLE PYBIND11_CMAKE_DIR
        OUTPUT_STRIP_TRAILING_WHITESPACE
        ERROR_QUIET
    )
    if(PYBIND11_CMAKE_DIR)
        find_package(pybind11 REQUIRED PATHS ${PYBIND11_CMAKE_DIR})
    else()
        message(FATAL_ERROR "pybind11 not found")
    endif()
endif()

# CUDA
if(VOXELVK_ENABLE_CUDA)
    find_package(CUDA REQUIRED)
    enable_language(CUDA)
    
    # Set CUDA architectures
    set(CMAKE_CUDA_ARCHITECTURES "70;75;80;86;89")
    
    message(STATUS "CUDA version: ${CUDA_VERSION}")
    message(STATUS "CUDA architectures: ${CMAKE_CUDA_ARCHITECTURES}")
endif()

# TensorRT (optional)
if(VOXELVK_ENABLE_TENSORRT)
    find_path(TENSORRT_INCLUDE_DIR NvInfer.h 
        HINTS 
        $ENV{TENSORRT_ROOT}/include
        /usr/include/x86_64-linux-gnu
        /usr/local/cuda/include
        /opt/tensorrt/include
    )
    
    find_library(TENSORRT_LIBRARY nvinfer 
        HINTS 
        $ENV{TENSORRT_ROOT}/lib
        /usr/lib/x86_64-linux-gnu
        /usr/local/cuda/lib64
        /opt/tensorrt/lib
    )
    
    find_library(TENSORRT_ONNX_PARSER nvonnxparser
        HINTS 
        $ENV{TENSORRT_ROOT}/lib
        /usr/lib/x86_64-linux-gnu
        /usr/local/cuda/lib64
        /opt/tensorrt/lib
    )
    
    if(TENSORRT_INCLUDE_DIR AND TENSORRT_LIBRARY AND TENSORRT_ONNX_PARSER)
        set(VOXELVK_HAS_TENSORRT ON)
        message(STATUS "TensorRT found: ${TENSORRT_LIBRARY}")
    else()
        set(VOXELVK_HAS_TENSORRT OFF)
        message(WARNING "TensorRT not found, AI inference features will be limited")
    endif()
endif()

# OpenMP
find_package(OpenMP)
if(OpenMP_CXX_FOUND)
    message(STATUS "OpenMP found")
endif()

# Threading
find_package(Threads REQUIRED)

# VMA (Vulkan Memory Allocator)
find_path(VMA_INCLUDE_DIR vk_mem_alloc.h
    HINTS 
    ${CMAKE_SOURCE_DIR}/external/VulkanMemoryAllocator/include
    /usr/local/include
    /usr/include
)

if(VMA_INCLUDE_DIR)
    set(VMA_TARGET VMA::VMA)
    add_library(VMA::VMA INTERFACE IMPORTED)
    target_include_directories(VMA::VMA INTERFACE ${VMA_INCLUDE_DIR})
    message(STATUS "VMA found: ${VMA_INCLUDE_DIR}")
else()
    message(FATAL_ERROR "VMA (Vulkan Memory Allocator) not found")
endif()

# zlib for compression

)

# Create main library
if(NOT VOXELVK_HEADLESS_ONLY)
    add_library(VoxelVK_Core STATIC 
        ${VOXELVK_CORE_SOURCES} 
        ${VOXELVK_AI_SOURCES}
        ${VOXELVK_CUDA_SOURCES}
    )
    
    target_include_directories(VoxelVK_Core PUBLIC 
        ${CMAKE_SOURCE_DIR}/src
        ${CMAKE_SOURCE_DIR}/src/ai
        ${Vulkan_INCLUDE_DIRS}
        ${Python3_INCLUDE_DIRS}
    )
    
    target_link_libraries(VoxelVK_Core PUBLIC
        ${Vulkan_LIBRARIES}
        ${VMA_TARGET}
        Threads::Threads
        ZLIB::ZLIB
        ${Python3_LIBRARIES}
    )
    
    # GLM
    if(glm_FOUND)
        target_link_libraries(VoxelVK_Core PUBLIC glm::glm)
    else()
        target_include_directories(VoxelVK_Core PUBLIC ${GLM_INCLUDE_DIR})
    endif()
    
    # OpenMP
    if(OpenMP_CXX_FOUND)
        target_link_libraries(VoxelVK_Core PUBLIC OpenMP::OpenMP_CXX)
    endif()
    
    # CUDA
    if(VOXELVK_ENABLE_CUDA)
        target_link_libraries(VoxelVK_Core PUBLIC ${CUDA_LIBRARIES})
        target_include_directories(VoxelVK_Core PUBLIC ${CUDA_INCLUDE_DIRS})
        set_property(TARGET VoxelVK_Core PROPERTY CUDA_SEPARABLE_COMPILATION ON)
    endif()
    
    # TensorRT
    if(VOXELVK_HAS_TENSORRT)
        target_include_directories(VoxelVK_Core PUBLIC ${TENSORRT_INCLUDE_DIR})
        target_link_libraries(VoxelVK_Core PUBLIC ${TENSORRT_LIBRARY} ${TENSORRT_ONNX_PARSER})
    endif()
    
    # Compile definitions
    target_compile_definitions(VoxelVK_Core PUBLIC
        $<$<BOOL:${VOXELVK_ENABLE_AI_GENERATION}>:VOXELVK_ENABLE_AI_GENERATION=1>
        $<$<BOOL:${VOXELVK_ENABLE_AI_TRAINING}>:VOXELVK_ENABLE_AI_TRAINING=1>
        $<$<BOOL:${VOXELVK_HAS_TENSORRT}>:VOXELVK_HAS_TENSORRT=1>
        $<$<BOOL:${VOXELVK_ENABLE_CUDA}>:VOXELVK_ENABLE_CUDA=1>
    )
    
    # Fix Vulkan struct initialization warnings
    target_compile_definitions(VoxelVK_Core PRIVATE
        VK_NO_PROTOTYPES=1
        VULKAN_HPP_NO_EXCEPTIONS=1
    )
    
    # Compiler-specific flags
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
        target_compile_options(VoxelVK_Core PRIVATE 
            -Wno-missing-field-initializers
            -Wno-unused-parameter
            -Wno-unused-variable
        )
    endif()
    
    set_target_properties(VoxelVK_Core PROPERTIES
        CXX_STANDARD 20
        CXX_STANDARD_REQUIRED ON
        POSITION_INDEPENDENT_CODE ON
    )
endif()

# Create Python module
if(pybind11_FOUND)
    pybind11_add_module(voxelvk_py ${VOXELVK_PYTHON_SOURCES})
    
    if(NOT VOXELVK_HEADLESS_ONLY)
        target_link_libraries(voxelvk_py PRIVATE VoxelVK_Core)
    endif()
    
    target_compile_definitions(voxelvk_py PRIVATE VERSION_INFO=${PROJECT_VERSION})
    
    # Set properties for Python module
    set_target_properties(voxelvk_py PROPERTIES
        CXX_VISIBILITY_PRESET hidden
        INTERPROCEDURAL_OPTIMIZATION ON
    )
endif()

# Create executables

    add_executable(VoxelVK_Trainer src/app/main_trainer.cpp)

# Testing
if(ENABLE_TESTS)
    # Find Google Test
    find_package(GTest QUIET)
    if(GTest_FOUND)
        enable_testing()
        
        # Core tests
        add_executable(VoxelVK_Core_Tests 
            tests/test_world_manager.cpp
            tests/test_chunk_manager.cpp
            tests/test_physics.cpp
        )
        target_link_libraries(VoxelVK_Core_Tests PRIVATE 
            VoxelVK_Core 
            GTest::gtest 
            GTest::gtest_main
        )
        
        # AI tests
        if(VOXELVK_ENABLE_AI_GENERATION)
            add_executable(VoxelVK_AI_Tests tests/test_ai_integration.cpp)
            target_link_libraries(VoxelVK_AI_Tests PRIVATE 
                VoxelVK_Core 
                GTest::gtest 
                GTest::gtest_main
            )
            target_include_directories(VoxelVK_AI_Tests PRIVATE ${CMAKE_SOURCE_DIR}/src)
        endif()
        
        # CUDA tests
        if(VOXELVK_ENABLE_CUDA)
            add_executable(VoxelVK_CUDA_Tests 
                tests/test_gpu_raycast.cpp
                tests/test_cuda_kernels.cpp
            )
            target_link_libraries(VoxelVK_CUDA_Tests PRIVATE 
                VoxelVK_Core 
                GTest::gtest 
                GTest::gtest_main
            )
        endif()
        
        # Add tests to CTest
        add_test(NAME VoxelVK_Core_Tests COMMAND VoxelVK_Core_Tests)
        if(VOXELVK_ENABLE_AI_GENERATION)
            add_test(NAME VoxelVK_AI_Tests COMMAND VoxelVK_AI_Tests)
        endif()
        if(VOXELVK_ENABLE_CUDA)
            add_test(NAME VoxelVK_CUDA_Tests COMMAND VoxelVK_CUDA_Tests)
        endif()
        
        # Set test properties
        set_tests_properties(VoxelVK_Core_Tests PROPERTIES TIMEOUT 60)
        if(VOXELVK_ENABLE_AI_GENERATION)
            set_tests_properties(VoxelVK_AI_Tests PROPERTIES TIMEOUT 120)
        endif()
        if(VOXELVK_ENABLE_CUDA)
            set_tests_properties(VoxelVK_CUDA_Tests PROPERTIES TIMEOUT 60)
        endif()
        
        message(STATUS "Tests enabled")
    else()
        message(WARNING "Google Test not found, tests disabled")
    endif()
endif()
